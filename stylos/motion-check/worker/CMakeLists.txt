cmake_minimum_required(VERSION 3.25)
# Keep user-wide MSBuild vcpkg integration from injecting dynamic image libraries.
list(APPEND CMAKE_VS_GLOBALS "VcpkgEnabled=false")
project(StylosRender64 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "StylosRender64 requires a Windows x64 toolchain")
endif()
find_package(ZLIB REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
file(GLOB_RECURSE renderer_sources CONFIGURE_DEPENDS rebuild/src/*.cpp rebuild/src/*.c)
add_library(stylos_frame_core STATIC ${renderer_sources})
target_compile_features(stylos_frame_core PUBLIC cxx_std_23 PRIVATE c_std_11)
target_include_directories(stylos_frame_core PUBLIC rebuild/include)
target_compile_definitions(stylos_frame_core PRIVATE RETAS_HAS_NATIVE_IMAGE_CODECS=1 RETAS_HAS_BATCH_SET=1)
set_source_files_properties(rebuild/src/io/batch_image_codec.cpp rebuild/src/io/batch_image_native.c
    PROPERTIES COMPILE_DEFINITIONS RETAS_RENDER_ALLOCATOR=1)
target_link_libraries(stylos_frame_core PUBLIC ZLIB::ZLIB PNG::PNG JPEG::JPEG bcrypt shlwapi shell32 ole32 gdi32)
add_executable(StylosRender64 main.cpp memory.cpp)
target_link_libraries(StylosRender64 PRIVATE stylos_frame_core advapi32 uuid psapi)
if(MSVC)
    set_property(TARGET stylos_frame_core StylosRender64 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(stylos_frame_core PRIVATE /bigobj /utf-8 /permissive- /Zc:__cplusplus /fp:strict)
    target_compile_options(StylosRender64 PRIVATE /utf-8 /Zc:__cplusplus)
else()
    target_compile_definitions(stylos_frame_core PRIVATE _WIN32_WINNT=0x0602)
    target_compile_options(stylos_frame_core PRIVATE -ffp-contract=off -fno-fast-math -Wa,-mbig-obj -ffunction-sections -fdata-sections)
    target_link_options(StylosRender64 PRIVATE -municode -static -Wl,--gc-sections -s)
endif()
install(TARGETS StylosRender64 RUNTIME DESTINATION .)
